# FileExplorer

~~~jsx
import React from 'react'
import {useState} from 'react';

const fileData = [
  {
    id: 1,
    name: 'README.md',
  },
  {
    id: 2,
    name: 'Documents',
    children: [
      {
        id: 3,
        name: 'Word.doc',
      },
      {
        id: 4,
        name: 'Powerpoint.ppt',
      },
    ],
  },
  {
    id: 5,
    name: 'Downloads',
    children: [
      {
        id: 6,
        name: 'unnamed.txt',
      },
      {
        id: 7,
        name: 'Misc',
        children: [
          {
            id: 8,
            name: 'foo.txt',
          },
          {
            id: 9,
            name: 'bar.txt',
          },
        ],
      },
    ],
  },
];

function sortItem(items) {
  return [...items].sort((a, b) => {
    const aIsDir = Array.isArray(a.children ?? []);
    const bIsDir = Array.isArray(b.children ?? []);
    if (aIsDir && !bIsDir) {
      return -1;
    }
    if (!aIsDir && bIsDir) {
      return 1;
    }
    return 0;
  });
}

function FileTreeItem({
  item,
  depth = 0
}) {
  if (item == null) {
    return null;
  }
  const isDirectory = item.children ? Array.isArray(item.children) : false;
  const [isExpand, setIsExpand] = useState(true);


  if (!isDirectory) {
    return (
      <div style={{ paddingLeft: depth * 16 + "px" }}>
        {item.name}
      </div>
    );
  }

  return (
    <div 
      onClick={() => setIsExpand(!isExpand)} 
      style={{ paddingLeft: depth * 16 + "px" }}
    >
      <span>{item.name}</span>
      {isExpand && <div>
        {sortItem(item.children).map(child => (
          <FileTreeItem 
            key={child.id} 
            item={child} 
            depth={depth+1} 
          />
        ))}
      </div>}
    </div>
  );
}
export default function FileExplorer() {
  const items = sortItem(fileData);

  console.log('>>> items', items);

  return (
    <div>
      {items.map(item => {
        return item && <FileTreeItem key={item.id} item={item} depth={0} />;
      })}
    </div>
  );
}
~~~
