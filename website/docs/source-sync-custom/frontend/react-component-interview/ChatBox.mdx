# ChatBox

~~~jsx
import React, {useState, useRef, useEffect} from "react";


const STORAGE_KEY = "prompt-playground-history";

function createRunId() {
  return crypto.randomUUID();
}

function sleep(ms) {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

function createMockResponse(prompt) {
  return `
You asked:

${prompt}

This is a mocked streaming response.

In a real app, this would come from a streaming API using fetch streams, Server-Sent Events, or WebSockets.

Main features implemented:
- streaming output
- cancellation
- run history
- local persistence
`.trim();
}

async function mockStream({ prompt, signal, onChunk }) {
  const response = createMockResponse(prompt);
  const words = response.split(" ");

  for (const word of words) {
    if (signal.aborted) {
      throw new DOMException("Aborted", "AbortError");
    }

    await sleep(80);

    if (signal.aborted) {
      throw new DOMException("Aborted", "AbortError");
    }

    onChunk(word + " ");
  }
}

export default function App() {
  const [prompt, setPrompt] = useState("");
  const [output, setOutput] = useState("");
  const [status, setStatus] = useState("idle");
  const [history, setHistory] = useState(() => {
    try {
      return JSON.parse(localStorage.getItem(STORAGE_KEY)) || [];
    } catch {
      return [];
    }
  });

  const abortRef = useRef(null);
  const runIdRef = useRef(null);

  const isRunning = status === "running";

  useEffect(() => {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(history));
  }, [history]);

  async function handleRun() {
    if (!prompt.trim() || isRunning) return;

    const runId = createRunId();
    const controller = new AbortController();

    runIdRef.current = runId;
    abortRef.current = controller;

    setOutput("");
    setStatus("running");

    const newRun = {
      id: runId,
      prompt,
      output: "",
      status: "running",
      createdAt: Date.now(),
    };

    try {
      let finalOutput = "";

      await mockStream({
        prompt,
        signal: controller.signal,
        onChunk: (chunk) => {
          if (runIdRef.current !== runId) return;

          finalOutput += chunk;
          setOutput(finalOutput);
        },
      });

      const completedRun = {
        ...newRun,
        output: finalOutput,
        status: "completed",
      };

      // wait for await to finish to run
      setHistory((prev) => [completedRun, ...prev]);
      setStatus("completed");
    } catch (error) {
      const cancelled = error.name === "AbortError";

      const cancelledRun = {
        ...newRun,
        output,
        status: cancelled ? "cancelled" : "error",
      };

      setHistory((prev) => [cancelledRun, ...prev]);
      setStatus(cancelled ? "cancelled" : "error");
    } finally {
      abortRef.current = null;
      runIdRef.current = null;
    }
  }

  function handleCancel() {
    abortRef.current?.abort();
  }

  function handleSelectRun(run) {
    if (isRunning) {
      abortRef.current?.abort();
    }

    setPrompt(run.prompt);
    setOutput(run.output);
    setStatus(run.status);
  }

  function handleClearHistory() {
    setHistory([]);
  }

  return (
    <div className="app">
      <h1>Prompt Playground</h1>

      <div className="layout">
        <main className="panel">
          <label>Prompt</label>

          <textarea
            value={prompt}
            disabled={isRunning}
            onChange={(e) => setPrompt(e.target.value)}
            placeholder="Enter your prompt..."
          />

          <div className="actions">
            <button onClick={handleRun} disabled={isRunning || !prompt.trim()}>
              Run
            </button>

            <button onClick={handleCancel} disabled={!isRunning}>
              Cancel
            </button>

            <span className="status">Status: {status}</span>
          </div>

          <h2>Output</h2>

          <pre className="output">
            {output || "No output yet."}
            {isRunning ? "▌" : ""}
          </pre>
        </main>

        <aside className="panel">
          <div className="history-header">
            <h2>History</h2>
            <button onClick={handleClearHistory}>Clear</button>
          </div>

          {history.length === 0 && <p>No history yet.</p>}

          {history.map((run) => (
            <div
              key={run.id}
              className="history-item"
              onClick={() => handleSelectRun(run)}
            >
              <strong>{run.prompt.slice(0, 50)}</strong>
              <div className="muted">
                {run.status} · {new Date(run.createdAt).toLocaleTimeString()}
              </div>
            </div>
          ))}
        </aside>
      </div>
    </div>
  );
}
~~~
