# Wordle

import "./styles.css";
import { useState } from "react";

const CORRECT_ANSWER = "SPEND";
const MAX_TRY = 5;
const WORD_LENGTH = CORRECT_ANSWER.length;

function validateWord(guess, answer) {
  const result = new Array(WORD_LENGTH).fill("missed");
  const remainingLetters = answer.split("");

  // First pass: exact matches.
  for (let i = 0; i < WORD_LENGTH; i++) {
    if (guess[i] === answer[i]) {
      result[i] = "correct";
      remainingLetters[i] = null;
    }
  }

  // Second pass: correct letter, wrong position.
  for (let i = 0; i < WORD_LENGTH; i++) {
    if (result[i] === "correct") continue;

    const matchIndex = remainingLetters.indexOf(guess[i]);

    if (matchIndex !== -1) {
      result[i] = "present";
      remainingLetters[matchIndex] = null;
    }
  }

  return result;
}

export default function App() {
  const [currentWord, setCurrentWord] = useState("");
  const [attempts, setAttempts] = useState([]);
  const [message, setMessage] = useState("");
  const [isGameOver, setIsGameOver] = useState(false);

  const handleSubmit = (event) => {
    event.preventDefault();

    if (isGameOver) return;

    if (currentWord.length !== WORD_LENGTH) {
      setMessage(`Please enter a ${WORD_LENGTH}-letter word.`);
      return;
    }

    const attempt = {
      word: currentWord,
      result: validateWord(currentWord, CORRECT_ANSWER),
    };

    const updatedAttempts = [...attempts, attempt];

    setAttempts(updatedAttempts);
    setCurrentWord("");

    if (currentWord === CORRECT_ANSWER) {
      setMessage("Won!");
      setIsGameOver(true);
    } else if (updatedAttempts.length >= MAX_TRY) {
      setMessage(`Game over! The word was ${CORRECT_ANSWER}.`);
      setIsGameOver(true);
    } else {
      setMessage("Not correct. Retry!");
    }
  };

  const handleChange = (event) => {
    if (isGameOver) return;

    const value = event.target.value
      .toUpperCase()
      .replace(/[^A-Z]/g, "")
      .slice(0, WORD_LENGTH);

    setCurrentWord(value);
  };

  const handleReset = () => {
    setCurrentWord("");
    setAttempts([]);
    setMessage("");
    setIsGameOver(false);
  };

  return (
    <main className="app">
      <h1>Wordle</h1>

      {message && (
        <div className="message" aria-live="polite">
          {message}
        </div>
      )}

      <div className="board">
        {Array.from({ length: MAX_TRY }, (_, rowIndex) => {
          const attempt = attempts[rowIndex];

          return (
            <div className="row" key={`row-${rowIndex}`}>
              {Array.from({ length: WORD_LENGTH }, (_, columnIndex) => {
                const letter = attempt?.word[columnIndex] ?? "";
                const status = attempt?.result[columnIndex] ?? "";

                return (
                  <div
                    className={`col ${status ? `col_${status}` : ""}`}
                    key={`cell-${rowIndex}-${columnIndex}`}
                  >
                    {letter}
                  </div>
                );
              })}
            </div>
          );
        })}
      </div>

      <form onSubmit={handleSubmit}>
        <input
          value={currentWord}
          disabled={isGameOver}
          onChange={handleChange}
          autoComplete="off"
          maxLength={WORD_LENGTH}
          aria-label="Enter a five-letter word"
        />

        <div className="actions">
          <button type="submit" disabled={isGameOver}>
            Submit
          </button>

          <button type="button" onClick={handleReset}>
            Reset
          </button>
        </div>
      </form>
    </main>
  );
}
