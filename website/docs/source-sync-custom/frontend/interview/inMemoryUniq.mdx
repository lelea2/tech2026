# inMemoryUniq

~~~typescript
function globalUniqBy<T, K>(
  items: T[],
  getKey: (item: T) => K
): T[] {
  const seen = new Set<K>();
  const result: T[] = [];

  for (const item of items) {
    const key = getKey(item);

    if (seen.has(key)) continue;

    seen.add(key);
    result.push(item);
  }

  return result;
}

const employees = [
  { id: 1, name: "Alice" },
  { id: 2, name: "Bob" },
  { id: 1, name: "Alice Updated" }
];

console.log(globalUniqBy(employees, employee => employee.id));
// [
//   { id: 1, name: "Alice" },
//   { id: 2, name: "Bob" }
// ]
~~~
