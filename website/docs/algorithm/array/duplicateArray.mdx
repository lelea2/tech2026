# duplicateArray

~~~javascript
/**
 * Determines if an array contains any duplicate elements.
 * @param {number[]} numbers - The array of numbers to check.
 * @return {boolean} - True if duplicates are found, false otherwise.
 */
// O(n) time complexity, O(n) space complexity
function containsDuplicate(numbers) {
  const seen = new Set();

  for (const num of numbers) {
    if (seen.has(num)) {
      return true;
    }
    seen.add(num);
  }

  return false;
}
~~~
