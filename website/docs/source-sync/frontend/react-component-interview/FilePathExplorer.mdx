# FilePathExplorer

~~~jsx
import React, { useState } from "react";

const HOME = "/home/user";

function isAbsolute(path) {
  return path.startsWith("/");
}

function splitPath(path) {
  return path.split("/").filter(Boolean);
}

function normalizePath(path) {
  const parts = splitPath(path);
  const stack = [];

  for (const part of parts) {
    if (part === ".") continue;

    if (part === "..") {
      if (stack.length === 0) return null;
      stack.pop(); // handle parent directory
    } else {
      stack.push(part); // handle regular directory
    }
  }

  return "/" + stack.join("/");
}

function joinPath(currentDir, newDir) {
  if (!newDir || newDir === ".") return normalizePath(currentDir);

  if (newDir === "~") {
    return HOME;
  }

  if (newDir.startsWith("~/")) {
    return normalizePath(HOME + "/" + newDir.slice(2));
  }

  if (isAbsolute(newDir)) {
    return normalizePath(newDir);
  }

  return normalizePath(currentDir.replace(/\/$/, "") + "/" + newDir);
}

function findLongestSoftLinkMatch(path, softLinks) {
  let bestMatch = null;

  for (const linkPath of Object.keys(softLinks)) {
    const normalizedLink = normalizePath(linkPath);

    if (!normalizedLink) continue;

    const isExactMatch = path === normalizedLink;
    const isPrefixMatch = path.startsWith(normalizedLink + "/");

    if (isExactMatch || isPrefixMatch) {
      if (!bestMatch || normalizedLink.length > bestMatch.length) {
        bestMatch = normalizedLink;
      }
    }
  }

  return bestMatch;
}

function resolveSoftLinks(path, softLinks) {
  let currentPath = path;
  const seen = new Set();

  while (true) {
    if (seen.has(currentPath)) {
      return null;
    }

    seen.add(currentPath);

    const match = findLongestSoftLinkMatch(currentPath, softLinks);

    if (!match) {
      return currentPath;
    }

    const target = softLinks[match];
    const remainder = currentPath.slice(match.length);

    let nextPath;

    if (isAbsolute(target)) {
      nextPath = normalizePath(target + remainder);
    } else {
      const parent = "/" + splitPath(match).slice(0, -1).join("/");
      nextPath = normalizePath(parent + "/" + target + remainder);
    }

    if (!nextPath) return null;

    currentPath = nextPath;
  }
}

function cd(currentDir, newDir, softLinks = {}) {
  const basePath = joinPath(currentDir, newDir);

  if (!basePath) return null;

  return resolveSoftLinks(basePath, softLinks);
}

const mockFileTree = {
  "/": ["foo", "home", "abc", "bcd"],
  "/foo": ["bar"],
  "/foo/bar": ["baz", "notes.txt"],
  "/foo/bar/baz": ["deep.txt"],
  "/home": ["user"],
  "/home/user": ["docs", "downloads"],
  "/home/user/docs": ["resume.pdf"],
  "/abc": ["baz"],
  "/abc/baz": ["from-abc.txt"],
  "/bcd": ["baz"],
  "/bcd/baz": ["from-bcd.txt"],
  "/xyz": ["final.txt"],
};

const defaultSoftLinks = {
  "/foo/bar": "/abc",
  "/abc": "/bcd",
  "/bcd/baz": "/xyz",
};

export default function FileExplorer() {
  const [currentDir, setCurrentDir] = useState("/");
  const [command, setCommand] = useState("");
  const [error, setError] = useState("");
  const [history, setHistory] = useState([]);

  const items = mockFileTree[currentDir] || [];

  function runCommand(nextCommand = command) {
    const nextPath = cd(currentDir, nextCommand, defaultSoftLinks);

    setHistory((prev) => [
      {
        from: currentDir,
        input: nextCommand,
        result: nextPath,
      },
      ...prev,
    ]);

    if (nextPath === null) {
      setError("Invalid path or symlink cycle detected.");
      return;
    }

    setCurrentDir(nextPath);
    setCommand("");
    setError("");
  }

  function handleSubmit(event) {
    event.preventDefault();
    runCommand();
  }

  function openItem(item) {
    runCommand(item);
  }

  function goUp() {
    runCommand("..");
  }

  return (
    <div style={styles.page}>
      <h2>File Explorer Path Resolver</h2>

      <div style={styles.panel}>
        <div style={styles.label}>Current Directory</div>
        <div style={styles.path}>{currentDir}</div>

        <form onSubmit={handleSubmit} style={styles.form}>
          <input
            value={command}
            onChange={(event) => setCommand(event.target.value)}
            placeholder='Try "foo/bar", "..", "~/docs", "/foo/bar/baz"'
            style={styles.input}
          />
          <button type="submit" style={styles.button}>
            cd
          </button>
        </form>

        {error && <div style={styles.error}>{error}</div>}
      </div>

      <div style={styles.panel}>
        <div style={styles.row}>
          <strong>Files</strong>
          <button onClick={goUp} style={styles.smallButton}>
            Up
          </button>
        </div>

        {items.length === 0 ? (
          <div style={styles.empty}>No files found for this path.</div>
        ) : (
          items.map((item) => (
            <button
              key={item}
              onClick={() => openItem(item)}
              style={styles.fileItem}
            >
              {item.includes(".") ? "📄" : "📁"} {item}
            </button>
          ))
        )}
      </div>

      <div style={styles.panel}>
        <strong>Soft Links</strong>

        <pre style={styles.pre}>
          {JSON.stringify(defaultSoftLinks, null, 2)}
        </pre>
      </div>

      <div style={styles.panel}>
        <strong>History</strong>

        {history.length === 0 ? (
          <div style={styles.empty}>No commands yet.</div>
        ) : (
          history.map((entry, index) => (
            <div key={index} style={styles.historyItem}>
              <div>
                <strong>From:</strong> {entry.from}
              </div>
              <div>
                <strong>cd:</strong> {entry.input}
              </div>
              <div>
                <strong>Result:</strong> {String(entry.result)}
              </div>
            </div>
          ))
        )}
      </div>
    </div>
  );
}
~~~
