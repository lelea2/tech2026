export function FormField({
  name,
  label,
  as: Component = "input",
  type = "text",
  options = [],
  parseValue,
  formatValue,
  getValueFromEvent,
  ...props
}) {
  const {
    values,
    errors,
    isSubmitting,
    setFieldValue,
  } = useFormContext();

  const rawValue = values[name];

  const displayValue = formatValue
    ? formatValue(rawValue)
    : rawValue ?? "";

  function normalizeValue(eventOrValue) {
    if (getValueFromEvent) {
      return getValueFromEvent(eventOrValue);
    }

    if (eventOrValue && eventOrValue.target) {
      const target = eventOrValue.target;

      if (target.type === "checkbox") {
        return target.checked;
      }

      if (target.type === "file") {
        return target.files;
      }

      return target.value;
    }

    return eventOrValue;
  }

  function handleChange(eventOrValue) {
    const nextValue = normalizeValue(eventOrValue);

    const parsedValue = parseValue
      ? parseValue(nextValue)
      : nextValue;

    setFieldValue(name, parsedValue);
  }

  function renderSelect() {
    return (
      <select
        id={name}
        name={name}
        value={displayValue}
        disabled={isSubmitting}
        onChange={handleChange}
        {...props}
      >
        {props.placeholder && (
          <option value="">{props.placeholder}</option>
        )}

        {options.map((option) => (
          <option
            key={option.value}
            value={option.value}
          >
            {option.label}
          </option>
        ))}
      </select>
    );
  }

  function renderTextarea() {
    return (
      <textarea
        id={name}
        name={name}
        value={displayValue}
        disabled={isSubmitting}
        onChange={handleChange}
        {...props}
      />
    );
  }

  function renderCheckbox() {
    return (
      <input
        id={name}
        name={name}
        type="checkbox"
        checked={Boolean(rawValue)}
        disabled={isSubmitting}
        onChange={handleChange}
        {...props}
      />
    );
  }

  function renderRadioGroup() {
    return (
      <div>
        {options.map((option) => (
          <label key={option.value} style={{ display: "block" }}>
            <input
              type="radio"
              name={name}
              value={option.value}
              checked={rawValue === option.value}
              disabled={isSubmitting}
              onChange={() => handleChange(option.value)}
            />
            {option.label}
          </label>
        ))}
      </div>
    );
  }

  function renderNativeInput() {
    return (
      <input
        id={name}
        name={name}
        type={type}
        value={type === "file" ? undefined : displayValue}
        disabled={isSubmitting}
        onChange={handleChange}
        {...props}
      />
    );
  }

  function renderCustomComponent() {
    return (
      <Component
        id={name}
        name={name}
        value={rawValue}
        disabled={isSubmitting}
        onChange={handleChange}
        error={errors[name]}
        {...props}
      />
    );
  }

  function renderField() {
    if (Component === "select") {
      return renderSelect();
    }

    if (Component === "textarea") {
      return renderTextarea();
    }

    if (type === "checkbox") {
      return renderCheckbox();
    }

    if (type === "radio") {
      return renderRadioGroup();
    }

    if (Component === "input") {
      return renderNativeInput();
    }

    return renderCustomComponent();
  }

  const fieldError = errors[name];

  return (
    <div style={{ marginBottom: 16 }}>
      {type === "checkbox" ? (
        <label>
          {renderField()} {label}
        </label>
      ) : (
        <>
          {label && (
            <label
              htmlFor={name}
              style={{ display: "block", marginBottom: 4 }}
            >
              {label}
            </label>
          )}

          {renderField()}
        </>
      )}

      {fieldError && (
        <p style={{ color: "red", marginTop: 4 }}>
          {fieldError}
        </p>
      )}
    </div>
  );
}

import React, {
  createContext,
  useContext,
  useMemo,
  useState,
} from "react";

/**
 * ============================================================
 * Form Context
 * ============================================================
 */

const FormContext = createContext(null);

function useFormContext() {
  const context = useContext(FormContext);

  if (!context) {
    throw new Error("Form components must be used inside <GenericForm>");
  }

  return context;
}

/**
 * ============================================================
 * GenericForm
 * ============================================================
 *
 * Responsibilities:
 * - own form values
 * - own errors
 * - own submit lifecycle
 * - expose setFieldValue to children
 * - run validation
 * - call onSubmit
 */

export function GenericForm({
  initialValues = {},
  validate,
  onSubmit,
  children,
}) {
  const [values, setValues] = useState(initialValues);
  const [errors, setErrors] = useState({});
  const [isSubmitting, setIsSubmitting] = useState(false);

  function setFieldValue(name, value) {
    setValues((prevValues) => ({
      ...prevValues,
      [name]: value,
    }));

    setErrors((prevErrors) => ({
      ...prevErrors,
      [name]: undefined,
    }));
  }

  function resetForm() {
    setValues(initialValues);
    setErrors({});
  }

  async function handleSubmit(event) {
    event.preventDefault();

    const validationErrors = validate ? validate(values) : {};
    setErrors(validationErrors);

    if (Object.keys(validationErrors).length > 0) {
      return;
    }

    try {
      setIsSubmitting(true);

      await onSubmit(values, {
        resetForm,
        setErrors,
        setFieldValue,
      });
    } catch (error) {
      setErrors((prevErrors) => ({
        ...prevErrors,
        submit: error.message || "Something went wrong",
      }));
    } finally {
      setIsSubmitting(false);
    }
  }

  const contextValue = useMemo(
    () => ({
      values,
      errors,
      isSubmitting,
      setFieldValue,
      setErrors,
      resetForm,
    }),
    [values, errors, isSubmitting]
  );

  return (
    <FormContext.Provider value={contextValue}>
      <form onSubmit={handleSubmit} noValidate>
        {children}

        {errors.submit && (
          <p style={{ color: "red" }}>{errors.submit}</p>
        )}
      </form>
    </FormContext.Provider>
  );
}

function UserForm() {
  // Initial values define the full shape of the form state. This prevents
  // uncontrolled inputs and makes reset behavior straightforward.
  const initialValues = {
    name: "",
    email: "",
    age: "",
    bio: "",
    role: "",
    startDate: "",
    contactMethod: "email",
    agree: false,
    resume: null,
  };

  // Validation is intentionally kept outside GenericForm so the reusable form
  // system is not coupled to one product's business rules.
  // Return an object where each key matches a field name.
  function validate(values) {
    const errors = {};

    if (!values.name.trim()) {
      errors.name = "Name is required";
    }

    if (!values.email.trim()) {
      errors.email = "Email is required";
    } else if (!values.email.includes("@")) {
      errors.email = "Email is invalid";
    }

    if (!values.age) {
      errors.age = "Age is required";
    } else if (values.age < 18) {
      errors.age = "Age must be at least 18";
    }

    if (!values.role) {
      errors.role = "Role is required";
    }

    if (!values.startDate) {
      errors.startDate = "Start date is required";
    }

    if (!values.agree) {
      errors.agree = "You must agree to continue";
    }

    return errors;
  }

  // GenericForm should call this only after validate returns no errors.
  // `helpers` lets the reusable form expose actions such as resetForm without
  // leaking its internal state implementation.
  async function handleSubmit(values, helpers) {
    console.log("Submitting values:", values);

    /**
     * If the form includes files, use FormData because File objects cannot be
     * sent correctly as plain JSON in a normal HTTP request body.
     */
    const formData = new FormData();

    Object.entries(values).forEach(([key, value]) => {
      // Keep file uploads as binary data; convert normal primitive values to
      // strings because FormData stores fields as strings or blobs.
      if (value instanceof File) {
        formData.append(key, value);
      } else {
        formData.append(key, String(value ?? ""));
      }
    });

    /**
     * Example request.
     * In an interview, this can be implemented with fetch, a GraphQL mutation,
     * or any API client. The important point is that GenericForm should not be
     * coupled to the transport layer.
     */
    const response = await fakeApiSubmit(values);

    if (!response.ok) {
      throw new Error("Failed to submit form");
    }

    // Reset after a successful submit so the UI returns to a clean state.
    helpers.resetForm();
  }

  return (
    // GenericForm receives the reusable contract: the initial state,
    // validation function, and submit function. Children describe the fields.
    <GenericForm
      initialValues={initialValues}
      validate={validate}
      onSubmit={handleSubmit}
    >
      {/* Basic controlled text input. */}
      <FormField
        name="name"
        label="Name"
        as="input"
        placeholder="Enter your name"
      />

      {/* Email uses native input semantics plus custom validation above. */}
      <FormField
        name="email"
        label="Email"
        as="input"
        type="email"
        placeholder="Enter your email"
      />

      {/* parseValue normalizes raw DOM input before storing it in form state. */}
      <FormField
        name="age"
        label="Age"
        as="input"
        type="number"
        parseValue={(value) => {
          if (value === "") return "";
          return Number(value);
        }}
      />

      {/* Textarea shows the same abstraction can support larger text fields. */}
      <FormField
        name="bio"
        label="Bio"
        as="textarea"
        rows={4}
        placeholder="Tell us about yourself"
      />

      {/* Select fields store one value from a known option list. */}
      <FormField
        name="role"
        label="Role"
        as="select"
        placeholder="Select a role"
        options={[
          { label: "User", value: "user" },
          { label: "Admin", value: "admin" },
          { label: "Manager", value: "manager" },
        ]}
      />

      {/* Custom components, such as DatePicker, can plug into the same field API. */}
      <FormField
        name="startDate"
        label="Start Date"
        as={DatePicker}
      />

      {/* Radio options share one field name and store the selected option value. */}
      <FormField
        name="contactMethod"
        label="Preferred Contact Method"
        type="radio"
        options={[
          { label: "Email", value: "email" },
          { label: "Phone", value: "phone" },
        ]}
      />

      {/* Checkbox fields usually store booleans instead of strings. */}
      <FormField
        name="agree"
        label="I agree to the terms"
        as="input"
        type="checkbox"
      />

      {/* File input stores the selected File so submit can append it to FormData. */}
      <FormField
        name="resume"
        label="Resume"
        as="input"
        type="file"
        parseValue={(files) => files?.[0] ?? null}
      />

      {/* SubmitButton can read GenericForm context to disable while submitting. */}
      <SubmitButton>Create User</SubmitButton>
    </GenericForm>
  );
}
