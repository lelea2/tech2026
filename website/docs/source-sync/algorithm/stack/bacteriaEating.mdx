# bacteriaEating

~~~javascript
// Start from the left. If the left bacterium is larger than the next one, it eats it and grows. 
// Otherwise, the larger right bacterium becomes the active bacterium.
function survivingBacteria(names, sizes) {
  const stack = [];

  for (let i = 0; i < names.length; i++) {
    const curr = {
      name: names[i],
      size: sizes[i]
    };

    // Any survivor to the left that is larger
    // can eat the current bacterium.
    if (
      stack.length > 0 &&
      stack[stack.length - 1].size > curr.size
    ) {
      continue;
    }

    // Current is larger than bacteria immediately
    // before it, but it CANNOT eat leftward.
    // So both survive.
    stack.push(curr);
  }

  return stack;
}

// Example usage:
const names = ['A', 'B', 'C', 'D', 'E'];
const sizes = [1, 3, 2, 5, 4];
console.log(survivingBacteria(names, sizes));
// Output: [{ name: 'B', size: 3 }, { name: 'D', size: 5 }] 

// Time complexity: O(n) - we traverse the list of bacteria once
// Space complexity: O(n) - in the worst case, all bacteria survive and are stored in the stack

~~~
