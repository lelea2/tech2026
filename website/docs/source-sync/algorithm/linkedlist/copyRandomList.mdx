# copyRandomList

~~~javascript
function copyRandomList (head) {
    if (!head) {
        return null;
    }

    const map = new Map();

    let curr = head;
    while (curr) {
        map.set(curr, new Node(curr.val, null, null));
        curr = curr.next;
    }
    curr = head;
    while (curr) {
        const copy = map.get(curr);
        copy.next = curr.next ? map.get(curr.next) : null;
        copy.random = curr.random ? map.get(curr.random) : null;
    }

    return map.get(head)
};

// Example usage:
const node1 = new Node(1);
const node2 = new Node(2);
const node3 = new Node(3);

node1.next = node2;
node2.next = node3;

node1.random = node3;
node2.random = node1;
node3.random = node2;

const copiedListHead = copyRandomList(node1);

// Verify the copied list
let curr = copiedListHead;
while (curr) {
    console.log(`Node value: ${curr.val}`);
    console.log(`Random points to: ${curr.random ? curr.random.val : 'null'}`);
    curr = curr.next;
}

// Time complexity: O(n) - we traverse the list twice, where n is the number of nodes
// Space complexity: O(n) - we use a hash map to store the mapping of original nodes to copied nodes    
~~~
