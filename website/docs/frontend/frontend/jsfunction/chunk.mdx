# chunk

~~~javascript
/**
 * @param {unknown[]} array
 * @param {number} [size=1]
 * @returns {unknown[][]}
 */
export default function chunk(array, size = 1) {
  const result = [];

  for (let i = 0; i < array.length; i += size) {
    result.push(array.slice(i, i + size));
  }

  return result;
}

// Example usage:
// chunk(['a', 'b', 'c', 'd']); // => [['a'], ['b'], ['c'], ['d']]
// chunk([1, 2, 3, 4], 2); // => [[1, 2], [3, 4]]
// chunk([1, 2, 3, 4], 3); // => [[1, 2, 3], [4]]
// chunk([]); // => []

~~~
